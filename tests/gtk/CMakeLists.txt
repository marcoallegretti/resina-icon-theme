cmake_minimum_required(VERSION 3.16)
project(resina-icon-gtk-tests LANGUAGES C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 gtk+-3.0)
pkg_check_modules(GTK4 gtk4)
if(GTK3_FOUND)
    add_executable(gtk3_probe gtk3_probe.c)
    target_include_directories(gtk3_probe PRIVATE ${GTK3_INCLUDE_DIRS})
    target_link_libraries(gtk3_probe PRIVATE ${GTK3_LIBRARIES})
endif()
if(GTK4_FOUND)
    add_executable(gtk4_probe gtk4_probe.c)
    target_include_directories(gtk4_probe PRIVATE ${GTK4_INCLUDE_DIRS})
    target_link_libraries(gtk4_probe PRIVATE ${GTK4_LIBRARIES})
endif()
